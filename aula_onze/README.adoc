        else {
            System.out.println("Você já está parado.");
        }
    }

    public void setQtdPaginas(int qtdPaginas) {
        this.qtdPaginas = qtdPaginas;

        this.preco = preco;

                ", autor='" + autor + '\'' +
                ", emprestado=" + emprestado +
                ", qtdPaginas=" + qtdPaginas +
                ", preco=" + preco +
                '}';
    }
    // verifica se o livro ja esta emprestado: se estiver, faz a devolucao;

            this.emprestado = true;

        }
    }

import java.util.ArrayList;
import java.util.List;
public class Biblioteca {
    private String nome;
    private List<Livro> livros;
    // construtores
    public Biblioteca(String nome, List<Livro> livros) {
        this.nome = nome;
        this.livros = livros;
    }
    public Biblioteca(String nome) {
        this.nome = nome;
        this.livros = new ArrayList<>();
    }
    public Biblioteca() {
        this.livros = new ArrayList<>();
    }
    // getters
    public String getNome() {
        return nome;
    }
    public List<Livro> getLivros() {
        return livros;
    }
    // setters
    public void setNome(String nome) {
        this.nome = nome;
    }
    public void setLivros(List<Livro> livros) {
        this.livros = livros;
    }
    public void adicionarLivro(Livro l) {
        this.livros.add(l);
    }
    public void imprimirLivroMaisCaro() {
        if (livros.isEmpty()) {
            System.out.println("A biblioteca nao possui livros cadastrados.");
            return;
        }
        Livro maisCaro = livros.get(0);
        for (Livro l : livros) {
            if (l.getPreco() > maisCaro.getPreco()) {
                maisCaro = l;
            }
        }
        System.out.println("Livro mais caro: " + maisCaro);
    }
    public double calcularMediaPreco() {
        if (livros.isEmpty()) {
            return 0.0;
        }
        double soma = 0.0;
        for (Livro l : livros) {
            soma += l.getPreco();
        }
        return soma / livros.size();
    }
    public int contarLivrosAutor(String nomeAutor) {
        int contador = 0;
        for (Livro l : livros) {
            if (l.getAutor() != null && l.getAutor().equalsIgnoreCase(nomeAutor)) {
                contador++;
            }
        }
        return contador;
    }
}

public class ContaBancaria {
    private String numeroConta;
    private double saldo;
    private String emailTitular;
    public ContaBancaria(String numeroConta, double saldoInicial, String emailTitular) {
        this.numeroConta = numeroConta;
        this.saldo = saldoInicial;
        this.emailTitular = emailTitular;
    }
    public void depositar(double valor) {
        if (valor > 0) {
            saldo += valor;
            System.out.println("Depósito de R$ " + valor +
                    " realizado na conta " + numeroConta);
        }
    }
    public void sacar(double valor, ServicoNotificacao notificacao) {
        if (valor > 0 && saldo >= valor) {
            saldo -= valor;
            System.out.println("Saque de R$ " + valor + " realizado com sucesso.");
            notificacao.enviarEmail(emailTitular, "Foi realizado um saque de R$ " + valor
                    + ". Saldo atual: R$ " + saldo);
        } else {
            System.out.println("Saldo insuficiente ou valor inválido para saque.");
        }
    }
    public double getSaldo() { return saldo; }
}
